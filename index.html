function doGet(e) { // 建立排他鎖，防止多人同時讀寫造成資料錯亂（最長等待 10 秒） var lock = LockService.getScriptLock(); try { lock.waitLock(10000); } catch (err) { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: "系統忙碌中（取得鎖定逾時），請稍後再試。" })).setMimeType(ContentService.MimeType.JSON); } try { var params = (e && e.parameter) ? e.parameter : {}; var action = String(params.action || "borrow").trim(); var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName("Equipment_List") || ss.getSheets()[0]; // 確保有資料可讀取 var lastRow = sheet.getLastRow(); if (lastRow < 2) { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: "表格內沒有找到設備資料（少於 2 行）。" })).setMimeType(ContentService.MimeType.JSON); } // 讀取範圍資料 (從第 1 行到最後一行，涵蓋欄位 A 到 E) var data = sheet.getRange(1, 1, lastRow, Math.max(sheet.getLastColumn(), 5)).getValues(); // 1. 取得所有已借出清單 if (action === "getBorrowedList") { var borrowedList = []; for (var i = 1; i < data.length; i++) { var currentStatus = String(data[i][2] || "").trim(); if (currentStatus === "借出中" || currentStatus === "已借出") { borrowedList.push({ id: String(data[i][0] || "").trim(), name: String(data[i][1] || "").trim(), brand: String(data[i][4] || "").trim() }); } } return ContentService.createTextOutput(JSON.stringify({ status: "success", data: borrowedList })) .setMimeType(ContentService.MimeType.JSON); } // 2. 借用與歸還邏輯 var equipmentId = params.id; if (!equipmentId) { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: "請提供設備 ID" })) .setMimeType(ContentService.MimeType.JSON); } var targetRow = -1; var eqName = ""; var eqBrand = ""; var currentStatus = ""; // 尋找對應的設備編號（忽略前後空白與大小寫差異） var searchId = String(equipmentId).trim().toLowerCase(); for (var i = 1; i < data.length; i++) { var rowId = String(data[i][0] || "").trim().toLowerCase(); if (rowId === searchId) { targetRow = i + 1; // 試算表列數從 1 開始，陣列從 0 開始，故 +1 eqName = data[i][1]; eqBrand = data[i][4] || ""; currentStatus = String(data[i][2] || "").trim(); break; } } if (targetRow === -1) { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: "找不到設備編號：" + equipmentId })).setMimeType(ContentService.MimeType.JSON); } // 執行借用動作 if (action === "borrow") { if (currentStatus === "可借用") { sheet.getRange(targetRow, 3).setValue("借出中"); return ContentService.createTextOutput(JSON.stringify({ status: "success", message: "借用成功！\n" + eqBrand + " " + eqName + "\n狀態已更新為：借出中" })).setMimeType(ContentService.MimeType.JSON); } else { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: "借用失敗！該設備目前狀態為：" + currentStatus })).setMimeType(ContentService.MimeType.JSON); } } // 執行歸還動作 else if (action === "return") { if (currentStatus === "借出中" || currentStatus === "已借出") { sheet.getRange(targetRow, 3).setValue("可借用"); return ContentService.createTextOutput(JSON.stringify({ status: "success", message: "歸還成功！\n" + eqBrand + " " + eqName + "\n狀態已更新為：可借用" })).setMimeType(ContentService.MimeType.JSON); } else { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: "歸還失敗！該設備目前狀態為：" + currentStatus + "（未在借出狀態）" })).setMimeType(ContentService.MimeType.JSON); } } else { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: "不支援的操作指令 (action)：" + action })).setMimeType(ContentService.MimeType.JSON); } } catch (error) { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: "伺服器發生內部錯誤：" + error.toString() })).setMimeType(ContentService.MimeType.JSON); } finally { // 確保無論成功 or 失敗，最後一定會釋放鎖定 lock.releaseLock(); } }
